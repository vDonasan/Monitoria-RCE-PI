Variáveis e Tipos de Dados

int · float · double · char

Em C todo dado precisa de um tipo declarado. O compilador usa isso para reservar o espaço correto na memória. Veja os 4 tipos primitivos mais usados:

Exemplo Completo
#include <stdio.h>

int main() {
    // Tipos inteiros
    int    idade    = 20;
    short  ano      = 2024;
    long   populacao= 8000000000L;

    // Tipos reais
    float  preco    = 19.99f;
    double pi       = 3.14159265358979;

    // Caractere
    char   inicial  = 'A';

    // Constante (não pode ser alterada)
    const int MAX = 100;

    printf("Idade: %d\n", idade);
    printf("Preco: %.2f\n", preco);
    printf("Pi:    %.5lf\n", pi);
    printf("Letra: %c\n", inicial);

    // Casting: forçar conversão de tipo
    int a = 7, b = 2;
    double resultado = (double)a / b; // 3.5 (não 3)
    printf("7/2 = %.1lf\n", resultado);

    return 0;
}

Estruturas de Controle

if · else · switch · for · while · do-while

Estruturas de controle determinam o fluxo do programa. Combinar condicionais com laços é a base de toda lógica de programação.

Exemplo — Todos os laços
#include <stdio.h>

int main() {
    // FOR: sabe quantas vezes vai repetir
    printf("Tabuada do 7:\n");
    for (int i = 1; i <= 10; i++) {
        printf("7 x %2d = %2d\n", i, 7*i);
    }

    // WHILE: enquanto condição for verdadeira
    int n = 1;
    printf("\nPotências de 2: ");
    while (n <= 1024) {
        printf("%d ", n);
        n *= 2;
    }

    // DO-WHILE: executa ao menos 1 vez
    int op;
    do {
        printf("\n\nMenu: 1-Jogar 2-Sair: ");
        scanf("%d", &op);
    } while (op != 1 && op != 2);

    // SWITCH: múltiplos casos
    switch (op) {
        case 1: printf("Jogando!\n"); break;
        case 2: printf("Saindo!\n");  break;
        default:printf("Inválido!\n");
    }
    return 0;
}

Funções e Recursão

protótipo · parâmetros · retorno · recursão

Funções dividem o problema em partes reutilizáveis. Recursão é quando uma função chama a si mesma — sempre precisa de um caso base para parar.

Exemplo — Funções + Recursão
#include <stdio.h>

// Protótipos (declaração antecipada)
float media(float a, float b, float c);
int   fatorial(int n);
int   fibonacci(int n);

int main() {
    printf("Média: %.2f\n", media(7.0f, 8.5f, 9.0f));
    printf("5! = %d\n",     fatorial(5));
    printf("Fib(8) = %d\n", fibonacci(8));
    return 0;
}

float media(float a, float b, float c) {
    return (a + b + c) / 3.0f;
}

// Recursão: n! = n * (n-1)!  | caso base: 0! = 1
int fatorial(int n) {
    if (n == 0) return 1;      // caso base
    return n * fatorial(n - 1); // chamada recursiva
}

// Recursão: fib(n) = fib(n-1) + fib(n-2)
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n-1) + fibonacci(n-2);
}

Ponteiros — Fundamentos

& · * · aritmética de ponteiros

Ponteiros armazenam endereços de memória. O operador & obtém o endereço; o operador * acessa o valor no endereço (desreferência).

Exemplo — Anatomia do ponteiro
#include <stdio.h>

int main() {
    int x = 42;
    int *p = &x;     // p aponta para x

    printf("Valor de x:       %d\n", x);
    printf("Endereço de x:    %p\n", (void*)&x);
    printf("Valor de p:       %p\n", (void*)p);
    printf("*p (conteúdo):   %d\n", *p);

    *p = 100;  // muda x através do ponteiro
    printf("x após *p=100: %d\n", x);

    // Aritmética de ponteiros em array
    int arr[] = {10, 20, 30, 40};
    int *ptr = arr;
    for (int i = 0; i < 4; i++) {
        printf("arr[%d] = %d\n", i, *(ptr + i));
    }
    return 0;
}

Alocação Dinâmica de Memória

malloc · calloc · realloc · free

Memória dinâmica é alocada no Heap em tempo de execução. Sempre valide o retorno de malloc e libere com free() para evitar memory leak.

Exemplo — Vetor dinâmico
#include <stdio.h>
#include <stdlib.h>

int main() {
    int n;
    printf("Quantos números? ");
    scanf("%d", &n);

    // Alocar no heap
    int *arr = (int*) malloc(n * sizeof(int));
    if (arr == NULL) { exit(1); } // erro

    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    // Expandir para o dobro
    int *temp = (int*) realloc(arr, 2*n * sizeof(int));
    if (temp != NULL) arr = temp;

    // Liberar memória
    free(arr);
    arr = NULL; // boa prática
    return 0;
}

Exemplo — Escrever e ler arquivo texto
#include <stdio.h>
#include <stdlib.h>

typedef struct { char nome[50]; int nota; } Aluno;

int main() {
    // Escrita
    FILE *arq = fopen("alunos.txt", "w");
    if (!arq) { perror("Erro"); return 1; }
    fprintf(arq, "Ana 95\n");
    fprintf(arq, "Bruno 72\n");
    fclose(arq);

    // Leitura
    arq = fopen("alunos.txt", "r");
    Aluno a;
    while (fscanf(arq, "%s %d", a.nome, &a.nota) == 2) {
        printf("%-10s: %d\n", a.nome, a.nota);
    }
    fclose(arq);
    return 0;
}

        while (i <= 100) { soma += i++; }
        System.out.println("Soma 1-100: " + soma);
    }

public class ContaBancaria {
    // Atributos (estado do objeto)
    private String titular;
    private double saldo;
    private int    numero;

    // Construtor
    public ContaBancaria(String titular, int numero) {
        this.titular = titular;
        this.numero  = numero;
        this.saldo   = 0.0;
    }

    // Métodos (comportamento)
    public void depositar(double valor) {
        if (valor > 0) saldo += valor;
    }

    public boolean sacar(double valor) {
        if (valor > 0 && valor <= saldo) {
            saldo -= valor;
            return true;
        }
        return false; // saldo insuficiente
    }

    public void exibirExtrato() {
        System.out.printf(
            "Conta: %d | Titular: %s | Saldo: R$ %.2f%n",
            numero, titular, saldo);
    }

    // Getters
    public double getSaldo()   { return saldo; }
    public String getTitular() { return titular; }
}
Usando a classe

Tabela Verdade — AND, OR, NOT

&& · || · ! · combinações

Operadores lógicos combinam condições booleanas. Entender a tabela verdade é essencial para escrever condicionais corretas.

Tabela Verdade Implementada em C
#include <stdio.h>

int main() {
    printf("A  B  | AND | OR  | NOT_A\n");
    printf("------+-----+-----+------\n");

    for (int A = 0; A <= 1; A++) {
        for (int B = 0; B <= 1; B++) {
            printf("%d  %d  |  %d  |  %d  |   %d\n",
                A, B,
                A && B,   // AND
                A || B,   // OR
                !A        // NOT A
            );
        }
    }

    // Expressão complexa com precedência
    int x = 5, y = 10, z = 3;
    int resultado = (x > z) && (y < 20) || (!(z == 3));
    printf("Resultado: %d\n", resultado);  // 1
    return 0;
}

02

Teste de Mesa

rastrear variáveis passo a passo

O teste de mesa é simular manualmente a execução de um código, anotando o valor de cada variável a cada passo. Essencial para depurar algoritmos antes de executá-los.

Código para rastrear
// Faça o teste de mesa deste código manualmente
int a = 1, b = 2, c = 0;

while (a < 5) {
    c = a + b;
    a = b;
    b = c;
    printf("%d ", c);
}

/*
Passo | a | b | c | Saída
------+---+---+---+-------
Início| 1 | 2 | 0 |
  1   | 2 | 3 | 3 | 3
  2   | 3 | 5 | 5 | 5
  3   | 5 | 8 | 8 | 8
Fim (a >= 5)
Saída: 3 5 8  → Fibonacci!
*/
